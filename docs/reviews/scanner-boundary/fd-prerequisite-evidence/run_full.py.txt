"""Private full-suite evidence; run only after clean signed candidate freeze."""
import hashlib
import json
import os
import subprocess
import time
import xml.etree.ElementTree as ET
import scanner_evidence as e

folder=e.HERE/'full-ci'
folder.mkdir(mode=0o700)
state=lambda:subprocess.check_output(['git','status','--porcelain=v1'],cwd=e.W,text=True)
assert state()=='', 'Full CI requires clean candidate'
head=subprocess.check_output(['git','rev-parse','HEAD'],cwd=e.W,text=True).strip()
sig=subprocess.check_output(['git','log','-1','--format=%G?'],cwd=e.W,text=True).strip()
assert sig=='G', 'Candidate signature must verify'
env=os.environ.copy()
env.update(PYTHONPATH=str(e.HERE),HZ_REQUIRE_TMUX='1',HYPOTHESIS_STORAGE_DIRECTORY=str(e.HERE/'hypothesis'),SCANNER_COLLECTION=str(folder/'collected.json'))
origin=subprocess.run([e.PYTHON,'-c','import sys, heterodyne.config.secret_scan as s; print(sys.version); print(sys.executable); print(s.__file__)'],cwd=e.W,env=env,text=True,capture_output=True,check=True).stdout
assert str(e.W/'src/heterodyne/config/secret_scan.py') in origin
(folder/'origin.txt').write_text(origin)
cmd=[e.PYTHON,'-m','pytest','-q','--ignore=tests/live','-p','scanner_probe']
start=time.monotonic()
with (folder/'collect.log').open('w') as log:
    collection=subprocess.run([*cmd,'--collect-only'],cwd=e.W,env=env,stdout=log,stderr=subprocess.STDOUT,timeout=180)
assert collection.returncode==0, 'Full collection failed'
pinned=json.loads((folder/'collected.json').read_text())
assert pinned and len(set(pinned))==len(pinned)
assert len(set(map(e.mask,pinned)))==len(pinned), 'Masked collection collision'
assert not any(n.startswith('tests/live/') for n in pinned)
(folder/'collection-pinned.json').write_text(json.dumps(pinned,indent=2)+'\n')
with (folder/'run.log').open('w') as log:
    actual=subprocess.run(['timeout','3600',*cmd,'--junitxml='+str(folder/'junit.xml')],cwd=e.W,env=env,stdout=log,stderr=subprocess.STDOUT)
(folder/'status.json').write_text(json.dumps({'actual_exit':actual.returncode})+'\n')
assert actual.returncode==0, 'Full suite failed; preserve evidence and hold'
assert json.loads((folder/'collected.json').read_text())==pinned
root=ET.fromstring((folder/'junit.xml').read_bytes())
assert root.tag=='testsuites' and root.findall('testsuite')
assert not root.findall('.//error') and not root.findall('.//failure')
nodes=[]; skips=[]
for case in root.findall('.//testcase'):
    ids=case.findall('./properties/property[@name="scanner_nodeid"]')
    assert len(ids)==1, 'Missing/duplicate full node identity'
    node=ids[0].attrib['value']; nodes.append(node)
    for skipped in case.findall('skipped'):
        skips.append({'node':node,'type':skipped.attrib.get('type'),'message':skipped.attrib.get('message'),'text':skipped.text})
assert len(nodes)==len(set(nodes)) and set(nodes)==set(pinned)
for suite in root.iter('testsuite'):
    assert int(suite.attrib['tests'])==len(suite.findall('testcase'))
    assert int(suite.attrib.get('failures',0))==0 and int(suite.attrib.get('errors',0))==0
    assert int(suite.attrib.get('skipped',0))==len(suite.findall('testcase/skipped'))
after=state(); assert after=='', 'Candidate changed during full CI'
assert subprocess.check_output(['git','rev-parse','HEAD'],cwd=e.W,text=True).strip()==head
result={'command':['timeout','3600',*cmd,'--junitxml='+str(folder/'junit.xml')],'head':head,'signature':sig,'before':'','after':after,'actual_exit':actual.returncode,'total':len(nodes),'passed':len(nodes)-len(skips),'skipped':len(skips),'duration_seconds':time.monotonic()-start,'collection_sha256':hashlib.sha256(json.dumps(pinned).encode()).hexdigest(),'log_sha256':hashlib.sha256((folder/'run.log').read_bytes()).hexdigest(),'origin':origin,'environment':{k:env[k] for k in ('PYTHONPATH','HZ_REQUIRE_TMUX','HYPOTHESIS_STORAGE_DIRECTORY')},'local_live_exclusion':'Explicit approved offline-only divergence from hosted unfiltered pytest; no live opt-in flags'}
(folder/'skips.json').write_text(json.dumps(skips,indent=2)+'\n')
(folder/'result.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps(result),flush=True)
