"""Disposable exact B1 scanner + ORIGINAL redactor; never committed or shipped."""
import hashlib
import json
import os
from pathlib import Path
import shutil
import subprocess
import time

from scanner_evidence import HERE, W, PYTHON, EXPECTED, verify

folder=HERE/'t2s5'
folder.mkdir(mode=0o700)
source=folder/'disposable-source'
source.mkdir(mode=0o700)
for name in ('src','tests'):
    shutil.copytree(W/name,source/name,ignore=shutil.ignore_patterns('__pycache__'))
shutil.copy2(W/'pyproject.toml',source/'pyproject.toml')
original=subprocess.check_output(['git','show','07637745fe75dc614109a0b8ba72d3489f262a97:src/heterodyne/admind/redact.py'],cwd=W)
(source/'src/heterodyne/admind/redact.py').write_bytes(original)
env=os.environ.copy()
env.update(PYTHONPATH=str(source/'src')+os.pathsep+str(HERE),HZ_REQUIRE_TMUX='1',HYPOTHESIS_STORAGE_DIRECTORY=str(HERE/'hypothesis-partial'),SCANNER_COLLECTION=str(folder/'collected.json'))
origin=subprocess.check_output([PYTHON,'-c','import sys,heterodyne.config.secret_scan as s,heterodyne.admind.redact as r; print(sys.version); print(s.__file__); print(r.__file__)'],cwd=source,env=env,text=True)
assert str(source/'src/heterodyne/config/secret_scan.py') in origin
assert str(source/'src/heterodyne/admind/redact.py') in origin
(folder/'origin.txt').write_text(origin)
cmd=[PYTHON,'-m','pytest','-q','-p','scanner_probe','tests/test_secret_scan_boundary.py','-k','overlapping_kinds']
t=time.monotonic()
with (folder/'collect.log').open('w') as f:
    c=subprocess.run([*cmd,'--collect-only'],cwd=source,env=env,stdout=f,stderr=subprocess.STDOUT,timeout=180)
assert c.returncode==0
nodes=json.loads((folder/'collected.json').read_text())
assert len(nodes)==7
(folder/'collection-pinned.json').write_text(json.dumps(nodes,indent=2)+'\n')
with (folder/'run.log').open('w') as f:
    actual=subprocess.run(['timeout','3600',*cmd,'--junitxml='+str(folder/'junit.xml')],cwd=source,env=env,stdout=f,stderr=subprocess.STDOUT)
(folder/'status.json').write_text(json.dumps({'actual_exit':actual.returncode})+'\n')
assert nodes==json.loads((folder/'collected.json').read_text())
result=verify((folder/'junit.xml').read_bytes(),nodes,actual.returncode,EXPECTED['t2s5']['failing_nodes'])
result.update(command=cmd,actual_exit=actual.returncode,source=str(source),origin=origin,duration_seconds=time.monotonic()-t,original_redactor_sha256=hashlib.sha256(original).hexdigest(),log_sha256=hashlib.sha256((folder/'run.log').read_bytes()).hexdigest(),collection_sha256=hashlib.sha256(json.dumps(nodes).encode()).hexdigest(),main_worktree_head=subprocess.check_output(['git','rev-parse','HEAD'],cwd=W,text=True).strip())
(folder/'result.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps(result))
