import json
from scanner_evidence import verify, HERE

xml=b'<testsuites><testsuite tests="1" failures="1" errors="0" skipped="0"><testcase><properties><property name="scanner_nodeid" value="tests/test_x.py::test_x" /></properties><failure message="assert False" /></testcase></testsuite></testsuites>'
node='tests/test_x.py::test_x'
assert verify(xml,[node],1,[node])=={'total':1,'failed':1,'passed':0}
probes={
 'invalid_root':(xml.replace(b'testsuites',b'notjunit'),[node],1,[node]),
 'false_suite_total':(xml.replace(b'tests="1"',b'tests="2"'),[node],1,[node]),
 'false_failure_total':(xml.replace(b'failures="1"',b'failures="0"'),[node],1,[node]),
 'malformed_xml':(b'<bad',[node],1,[node]),
 'wrong_exit_zero':(xml,[node],0,[node]),
 'timeout_exit':(xml,[node],124,[node]),
 'missing_executable':(xml,[node],127,[node]),
 'nonassertion':(xml.replace(b'assert False',b'RuntimeError: broken'),[node],1,[node]),
 'error_instead_failure':(xml.replace(b'<failure ',b'<error '),[node],1,[node]),
 'skip':(xml.replace(b'<failure message="assert False" />',b'<skipped />'),[node],1,[node]),
 'missing_node':(xml,[node,'tests/test_x.py::test_y'],1,[node]),
 'duplicate_collection':(xml,[node,node],1,[node]),
 'unexpected_case':(xml,['tests/test_x.py::test_y'],1,[node]),
 'unexpected_failure':(xml,[node],1,['tests/test_x.py::test_y']),
 'empty_collection':(xml,[],1,[node]),
 'missing_identity':(xml.replace(b'scanner_nodeid',b'other'),[node],1,[node]),
 'setup_suite_error':(xml.replace(b'errors="0"',b'errors="1"'),[node],1,[node]),
 'masked_collision':(xml,['tests/a.py::test[aabbccddeeff0011]','tests/a.py::test[1122334455667788]'],1,[node]),
}
case=xml[xml.index(b'<testcase>'):xml.index(b'</testcase>')+len(b'</testcase>')]
probes['duplicate_testcase']=(xml.replace(case,case+case),[node],1,[node])
for name,args in probes.items():
 try:
  verify(*args)
 except Exception:
  pass
 else:
  raise AssertionError('accepted invalid '+name)
for msg in ('AssertionError: mismatch','Failed: DID NOT RAISE <class ValueError>'):
 assert verify(xml.replace(b'assert False',msg.replace('<','&lt;').replace('>','&gt;').encode()),[node],1,[node])['failed']==1
(HERE/'verifier-probes.json').write_text(json.dumps({'negative_probes':sorted(probes),'rejected':len(probes),'valid_assertion_forms':3,'stale_report_guard':'run creates exclusive per-stage directory before collection or invocation'},indent=2)+'\n')
print(len(probes),'negative verifier probes rejected; 3 assertion forms accepted')
